function howManyTimes(time1,time2){
  let start = new Date(time1.replace(" ","T"));
  let end = new Date(time2.replace(" ","T"));
  let stuki = 0;
  for(let time = start;time < end;time.setTime(time.getTime() + 1000))
  {
     let mins = time.getMinutes();
     let secs = time.getSeconds();
     let hours = time.getHours();
     if(mins == 30 && secs == 0)
     {
       stuki++;
       continue;
     }
     if(hours <= 12 && hours != 0)
     {
        if(mins == 0 && secs < hours)
        {
            stuki++;
        }
     }
     else
     {  
        if(hours == 0)
        {
          hours = 12;
        }
        else
        {
          hours -= 12;
        }
        if(mins == 0 && secs < hours)
        {
            stuki++;
        }
     }
  }
  return stuki;
}
