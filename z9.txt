function solution(str){
   let letters = str.split('');
   let res = [];
   if(letters.length % 2 == 0)
     {
       for(let i = 0;i < letters.length;i += 2)
         {
           
           let t = letters[i] + letters[i+1];
           res.push(t);
         }
     }
   else
     {
       for(let i = 0;i < letters.length;i += 2)
         {
           let t;
           if(i == letters.length - 1)
             {
               t = letters[i] + '_';
             }
           else
             {
               t = letters[i] + letters[i+1];
             }
           res.push(t);
         }
     }
  return res;
}
