function bingo(ticket, win){
  let countWin = 0;
  for(let i = 0;i < ticket.length;i++)
    {
      let str = ticket[i][0];
      let num = ticket[i][1];
      for(let j = 0;j < str.length;j++)
        {
          let char = str.charCodeAt(j);
          if(char == num)
            {
              countWin++;
              break;
            }
        }
    }
  let res;
  if(countWin >= win)
    {
      res = 'Winner!';
    }
  else
    {
      res = 'Loser!';
    }
  return res;
}
